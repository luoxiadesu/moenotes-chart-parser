cmake_minimum_required(VERSION 3.16)
project(moenotes-consumer LANGUAGES C CXX)
find_package(moenotes-chart-parser 0.1.0 CONFIG REQUIRED)
add_executable(consumer ../../examples/basic.c)
target_link_libraries(consumer PRIVATE moenotes::chart_parser)
add_executable(consumer-cpp cpp.cpp)
target_compile_features(consumer-cpp PRIVATE cxx_std_17)
target_link_libraries(consumer-cpp PRIVATE moenotes::chart_parser)
enable_testing()
add_test(NAME installed-c-consumer COMMAND consumer)
add_test(NAME installed-cpp-consumer COMMAND consumer-cpp)
